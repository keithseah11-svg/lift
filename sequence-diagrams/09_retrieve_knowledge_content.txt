title UC9 - Retrieve Knowledge Content (ECB)

actor User
boundary ":CommunicationUI" as CommunicationUI
control ":CommunicationController" as CommunicationController
control ":AccessController" as AccessController
entity ":KnowledgeVault" as KnowledgeVault
entity ":KnowledgeContent" as KnowledgeContent

User->CommunicationUI: open knowledge vault
activate CommunicationUI
CommunicationUI->CommunicationController: retrieveKnowledgeContent()
activate CommunicationController
CommunicationController->AccessController: checkAccess(userID, "viewKnowledge")
activate AccessController
AccessController-->CommunicationController: granted
deactivate AccessController
CommunicationController->KnowledgeVault: get contents
activate KnowledgeVault
loop for each KnowledgeContent in vault
KnowledgeVault->KnowledgeContent: get title, contentType, uploadDate
activate KnowledgeContent
KnowledgeContent-->KnowledgeVault: summary
deactivate KnowledgeContent
end
KnowledgeVault-->CommunicationController: content list
deactivate KnowledgeVault
CommunicationController-->CommunicationUI: content list
deactivate CommunicationController
CommunicationUI-->User: display content list
User->CommunicationUI: select selectedKnowledgeContent
CommunicationUI->CommunicationController: retrieveKnowledgeContent(contentID)
activate CommunicationController
CommunicationController->KnowledgeContent: get contentDescription
activate KnowledgeContent
KnowledgeContent-->CommunicationController: full content
deactivate KnowledgeContent
CommunicationController-->CommunicationUI: full content
deactivate CommunicationController
CommunicationUI-->User: display content
deactivate CommunicationUI
