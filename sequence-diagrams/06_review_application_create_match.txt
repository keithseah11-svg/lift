title UC6 - Review Application & Create Match (ECB)

actor ExistingHawker
boundary ":ApprenticeshipUI" as ApprenticeshipUI
control ":ApprenticeshipController" as ApprenticeshipController
control ":AccessController" as AccessController
entity ":Application" as Application
entity ":Match" as Match
entity ":Apprenticeship" as Apprenticeship
entity ":ApprenticeshipPhase" as ApprenticeshipPhase
control ":DataManager\n<<singleton>>" as DataManager

ExistingHawker->ApprenticeshipUI: open selectedApplication, choose decision
activate ApprenticeshipUI
ApprenticeshipUI->ApprenticeshipController: updateApplicationStatus(applicationID, status)
activate ApprenticeshipController
ApprenticeshipController->AccessController: checkAccess(userID, "updateApplicationStatus")
activate AccessController
AccessController-->ApprenticeshipController: granted
deactivate AccessController
ApprenticeshipController->Application: set applicationStatus = status
activate Application
Application-->ApprenticeshipController: updated
deactivate Application

alt status = "Accepted"
ApprenticeshipController->ApprenticeshipController: createMatch(applicationID)
activate ApprenticeshipController
ApprenticeshipController->*Match: <<create>> (matchDate, matchStatus = "Matched")
ApprenticeshipController->*Apprenticeship: <<create>> (startDate, expectedEndDate,\napprenticeshipStatus = "Active", overallProgress = 0)
loop for each training phase
ApprenticeshipController->*ApprenticeshipPhase: <<create>> (phaseName, sequenceNumber,\nphaseStatus = "Not Started")
end
deactivate ApprenticeshipController
ApprenticeshipController->DataManager: saveData()
activate DataManager
DataManager-->ApprenticeshipController: saved
deactivate DataManager
ApprenticeshipController-->ApprenticeshipUI: match & apprenticeship created
ApprenticeshipUI-->ExistingHawker: show "applicant matched"
else status = "Rejected"
ApprenticeshipController->DataManager: saveData()
activate DataManager
DataManager-->ApprenticeshipController: saved
deactivate DataManager
ApprenticeshipController-->ApprenticeshipUI: application rejected
ApprenticeshipUI-->ExistingHawker: show "application rejected"
end
deactivate ApprenticeshipController
deactivate ApprenticeshipUI
