title UC4 - Calculate Stall Viability (ECB)

actor AspiringHawker
boundary ":StallUI" as StallUI
control ":StallController" as StallController
entity ":Stall" as Stall
entity ":StallListing" as StallListing
entity ":ViabilityEstimate" as ViabilityEstimate
control ":DataManager\n<<singleton>>" as DataManager

AspiringHawker->StallUI: request viability for selectedStall
activate StallUI
StallUI->StallController: calculateViability(stallID)
activate StallController
StallController->Stall: get monthlyRent, location, cuisineType, yearsOpen
activate Stall
Stall-->StallController: stall data
deactivate Stall
StallController->StallListing: get listing data
activate StallListing
StallListing-->StallController: monthlyRent, minimumTrainingPeriod
deactivate StallListing
StallController->StallController: compute estimatedRevenue,\nestimatedExpenses, estimatedProfit
StallController->*ViabilityEstimate: <<create>> (estimatedRevenue, estimatedExpenses,\nestimatedProfit, calculationDate)
StallController->DataManager: saveData()
activate DataManager
DataManager-->StallController: saved
deactivate DataManager
StallController-->StallUI: viability estimate
deactivate StallController
StallUI-->AspiringHawker: display revenue / expenses / profit
deactivate StallUI
