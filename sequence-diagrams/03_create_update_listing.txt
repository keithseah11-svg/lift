title UC3 - Create / Update Stall Listing (ECB)

actor ExistingHawker
boundary ":StallUI" as StallUI
control ":StallController" as StallController
control ":AccessController" as AccessController
control ":DataManager\n<<singleton>>" as DataManager
entity ":Stall" as Stall
entity ":StallListingList\n<<container>>" as StallListingList
entity ":StallListing" as StallListing
entity ":CandidateRequirement" as CandidateRequirement

group Create listing
ExistingHawker->StallUI: fill in listingFormData
activate StallUI
StallUI->StallController: createListing(listingFormData)
activate StallController
StallController->AccessController: checkAccess(userID, "createListing")
activate AccessController
AccessController-->StallController: result
deactivate AccessController
alt access denied (not verified ExistingHawker)
StallController-->StallUI: access denied
StallUI-->ExistingHawker: show "only verified hawkers can list stalls"
else access granted
StallController->Stall: get stall (stallID)
activate Stall
Stall-->StallController: stall
deactivate Stall
StallController->*StallListing: <<create>> (cuisineType, location, monthlyRent,\nyearsOpen, retirementReason, minimumTrainingPeriod)
loop for each requirement in listingFormData
StallController->*CandidateRequirement: <<create>> (requirementType, description, mandatoryStatus)
end
StallController->StallListingList: addListing(listing)
activate StallListingList
StallListingList-->StallController: ok
deactivate StallListingList
StallController->DataManager: getInstance().saveData()
activate DataManager
DataManager-->StallController: saved
deactivate DataManager
StallController-->StallUI: listing created (listingID)
StallUI-->ExistingHawker: show confirmation
end
deactivate StallController
deactivate StallUI
end

group Update listing
ExistingHawker->StallUI: edit listingFormData
activate StallUI
StallUI->StallController: updateListing(listingID, listingFormData)
activate StallController
StallController->AccessController: checkAccess(userID, "updateListing")
activate AccessController
AccessController-->StallController: granted
deactivate AccessController
StallController->StallListing: update fields / listingStatus
activate StallListing
StallListing-->StallController: updated
deactivate StallListing
StallController->DataManager: saveData()
activate DataManager
DataManager-->StallController: saved
deactivate DataManager
StallController-->StallUI: listing updated
StallUI-->ExistingHawker: show updated listing
deactivate StallController
deactivate StallUI
end
